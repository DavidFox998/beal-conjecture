\section{Gap-3 $A^4+B^4=(B+3)^{13}$ toward Baker $B_0=10^6$}

\begin{definition}[Linear form $\Lambda_{\mathrm{gap3}}$]
For $A,B\in\mathbb{N}$, $B>0$, on $A^4+B^4=(B+3)^{13}$,
\[
\Lambda_{\mathrm{gap3}} := 4\log A - 13\log(B+3)
= \log\bigl(1+B^4/A^4\bigr),
\qquad |\Lambda|\le B^4/A^4.
\]
Here $B<A$ via \texttt{eq\_implies\_A\_gt\_B} and
$A\approx(B+3)^{13/4}$, not $A<B+3$. Also $A^4<(B+3)^{13}$,
$1<B+3$, and $\log A < \tfrac{13}{4}\log(B+3)$.
\end{definition}

\begin{definition}[Matveev 2000 Thm.~1.4 $n=2$ data $hGen$]
On a gap-3 solution one takes $\alpha_1=A$, $\alpha_2=B+3$
(not the Nat $B$), $b_1=4$, $b_2=-13$, $D=1$, $\kappa=1$,
$B_{\mathrm{coeff}}=\max|b_i|=13$,
$A_1=\log\alpha_1$, $A_2=\log\alpha_2$,
\[
B0_{\mathrm{term}}
= \log 13 + \log(\log 13) + \log\alpha_1 + \log\alpha_2.
\]
The displayed constants are
$C_1^{\mathrm{floor}}=143186215390$, $30^6=729000000$,
$height_{B0}=C_1^{\mathrm{floor}}\cdot 30^6=104382751019310000000$,
$C_{\mathrm{exp}}= -height_{B0}\log height_{B0}$, and
$0<\exp(C_{\mathrm{exp}})<10^{-12}$.
The general hypothesis is
\[
hGen:\quad
\forall\,\alpha_1,\alpha_2\in\mathbb{R},\
1<\alpha_1,\ 1<\alpha_2,\quad
\bigl|4\log\alpha_1-13\log\alpha_2\bigr|
> \exp\bigl(-C_1^{\mathrm{floor}} A_1 A_2 B0_{\mathrm{term}}\bigr).
\]
Nat $B=1$ occurs, so $1<B$ is the wrong hyp; one uses $1<\alpha_2=B+3$.
Mathlib~4.12 has no Matveev/Baker/Bugeaud linear-forms theorem;
$hGen$ is an uninhabited \texttt{def Prop} in
\texttt{beal-level-26-foundations} $4bd15bd$.
\end{definition}

\begin{definition}[Bugeaud LLL $hLLL$]
\[
hLLL:\quad
hGen \to
\bigl(\forall A,B\in\mathbb{N},\
A^4+B^4=(B+3)^{13},\ 0<B \to B\le B_0\bigr),
\qquad B_0=10^6.
\]
Raw Matveev versus $|\Lambda|\le B^4/A^4$ only rules out astronomically
large $B$; $B\le 10^6$ needs lattice reduction, which is not in Mathlib.
\end{definition}

\begin{lemma}[Product bound on the $B\le B_0$ range, proved in $4bd15bd$]
\label{lem:product}
If $B\le 10^6$ (equivalently $B+3\le height_{B0}$) then
\[
C_1^{\mathrm{floor}}
\log A\cdot\log(B+3)
\bigl(\log 13+\log(\log 13)+\log A+\log(B+3)\bigr)
\le height_{B0}\log height_{B0},
\]
via $\log 13<3$, $\log(\log 13)<1$, $\log height_{B0}<63$, and
$\log A<\tfrac{13}{4}\log(B+3)$.
Thus $hGen$ gives $|\Lambda|>\exp(C_{\mathrm{exp}})$ on the $B\le B_0$
range (\texttt{matveev\_gap3\_lower\_of\_general\_of\_B\_le\_B0}).
The unrestricted $\forall$-solutions product is not a theorem:
the left-hand side is $O((\log(B+3))^3)$ while the right-hand side is
the constant $\sim 5\times 10^{21}$.
\end{lemma}

\begin{theorem}[v25.0.0 $B_0$-unconditional, assuming $hGen$ and $hLLL$]
\label{thm:v25}
Assume $hGen$ and $hLLL$. Then
\[
\forall B\in\mathbb{N},\quad
\neg\exists A\in\mathbb{N},\quad
A^4+B^4=(B+3)^{13}.
\]
In particular gap-3 of Beal $x^4+y^4=z^{13}$ with $z-y=3$ has no
solutions. This is $B\le 10^6$ unconditional only after both
hypotheses are inhabited.
\end{theorem}

\begin{proof}
From $hLLL$ and $hGen$, the monorepo packaging
\texttt{baker\_bound\_gap3\_of\_LLL}~\cite{beal_level26_v25_0_0}
inhabits \texttt{baker\_bound\_gap3}: there is no gap-3 solution with
$B>10^6$.
Solutions with $B\le 10^6$ are already excluded by the v24.4.0 residue
census \texttt{baker\_le\_B0\_gap3}~\cite{beal_level26_v24_4_0}.
Therefore \texttt{baker\_conditional\_gap3\_full} (equivalently
\texttt{v25\_of\_hGen\_hLLL}) gives $\forall B,\ \neg\exists A$.

On the $B\le 10^6$ range, Lemma~\ref{lem:product} plus $hGen$ also
gives $|\Lambda|>\exp(C_{\mathrm{exp}})$. Combined with the closed
upper bound $|\Lambda|\le B^4/A^4$ this is the v24.4.0 conditional
$B^4(1+\exp(C_{\mathrm{exp}}))>\exp(C_{\mathrm{exp}})(B+3)^{13}$.
That inequality is not what kills $B\le 10^6$; the census does.
\end{proof}
